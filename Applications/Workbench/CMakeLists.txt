cmake_minimum_required(VERSION 3.22)
project(KartikayWorkbench LANGUAGES CXX)
add_executable(kit_processor processor.cxx)
target_compile_features(kit_processor PRIVATE cxx_std_17)
set_target_properties(kit_processor PROPERTIES CXX_EXTENSIONS OFF)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME WorkbenchOperations COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_processor.py $<TARGET_FILE:kit_processor>)
install(TARGETS kit_processor RUNTIME DESTINATION bin)
